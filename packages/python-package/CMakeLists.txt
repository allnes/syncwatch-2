cmake_minimum_required(VERSION 3.30)
project(SyncWatchPythonPackage LANGUAGES C)
add_library(syncwatch_python_package STATIC package.c paths.c digest.c python_paths.c)
if(WIN32)
    target_sources(syncwatch_python_package PRIVATE windows.c)
    target_compile_definitions(syncwatch_python_package PRIVATE _WIN32_WINNT=0x0A00)
else()
    target_sources(syncwatch_python_package PRIVATE posix.c)
endif()
target_compile_features(syncwatch_python_package PUBLIC c_std_11)
set_property(TARGET syncwatch_python_package PROPERTY POSITION_INDEPENDENT_CODE ON)
target_include_directories(syncwatch_python_package PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../python-startup")
target_include_directories(syncwatch_python_package SYSTEM PRIVATE "${SYNCWATCH_HASH}/include")
add_library(syncwatch_hacl STATIC IMPORTED)
if(WIN32)
    set_property(TARGET syncwatch_hacl PROPERTY IMPORTED_LOCATION "${SYNCWATCH_HASH}/lib/syncwatch_hacl.lib")
else()
    set_property(TARGET syncwatch_hacl PROPERTY IMPORTED_LOCATION "${SYNCWATCH_HASH}/lib/libsyncwatch_hacl.a")
endif()
target_link_libraries(syncwatch_python_package PUBLIC syncwatch_hacl)
if(MSVC)
    target_compile_options(syncwatch_python_package PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(syncwatch_python_package PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
