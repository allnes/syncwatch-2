cmake_minimum_required(VERSION 3.30)
project(SyncWatchHelper C)
add_executable(syncwatch-helper main.c ../python-startup/startup.c)
target_compile_features(syncwatch-helper PRIVATE c_std_11)
target_include_directories(syncwatch-helper PRIVATE ../python-startup)
target_include_directories(syncwatch-helper SYSTEM PRIVATE "${SYNCWATCH_RUNTIME}/include")
if(WIN32)
    target_link_libraries(syncwatch-helper PRIVATE "${SYNCWATCH_RUNTIME}/lib/python312.lib")
elseif(APPLE)
    target_link_libraries(syncwatch-helper PRIVATE "${SYNCWATCH_RUNTIME}/lib/libpython3.12.dylib")
    set_target_properties(syncwatch-helper PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/../lib")
else()
    target_link_libraries(syncwatch-helper PRIVATE "${SYNCWATCH_RUNTIME}/lib/libpython3.12.so.1.0")
    set_target_properties(syncwatch-helper PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
endif()
if(MSVC)
    target_compile_options(syncwatch-helper PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(syncwatch-helper PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
