cmake_minimum_required(VERSION 3.30)
project(SyncWatchUnicodeTypesProbe LANGUAGES C)
add_executable(unicode-probe probe.c)
target_compile_features(unicode-probe PRIVATE c_std_11)
set_target_properties(unicode-probe PROPERTIES C_EXTENSIONS OFF)
target_include_directories(unicode-probe PRIVATE "${SYNCWATCH_FIXTURE}")
if(MSVC)
    target_compile_options(unicode-probe PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(unicode-probe PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/probe-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:unicode-probe>")
