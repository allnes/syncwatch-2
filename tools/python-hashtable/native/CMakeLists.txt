cmake_minimum_required(VERSION 3.30)
project(SyncWatchHashtableProbe LANGUAGES C)
add_executable(hashtable-probe probe.c)
target_compile_features(hashtable-probe PRIVATE c_std_11)
set_target_properties(hashtable-probe PROPERTIES C_EXTENSIONS OFF)
target_include_directories(hashtable-probe PRIVATE "${SYNCWATCH_FIXTURE}")
if(MSVC)
    target_compile_options(hashtable-probe PRIVATE /W4 /WX /utf-8 /UNDEBUG)
else()
    target_compile_options(hashtable-probe PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/probe-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:hashtable-probe>")
