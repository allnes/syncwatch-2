cmake_minimum_required(VERSION 3.30)
project(SyncWatchSipHashProbe LANGUAGES C)
add_executable(siphash-probe probe.c)
target_compile_features(siphash-probe PRIVATE c_std_11)
set_target_properties(siphash-probe PROPERTIES C_EXTENSIONS OFF)
target_include_directories(siphash-probe PRIVATE "${SYNCWATCH_FIXTURE}")
if(MSVC)
    target_compile_options(siphash-probe PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(siphash-probe PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/probe-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:siphash-probe>")
