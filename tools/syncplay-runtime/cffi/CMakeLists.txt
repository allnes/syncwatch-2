cmake_minimum_required(VERSION 3.30)
project(SyncWatchCffi LANGUAGES C)

foreach(name CFFI_SOURCE PYTHON_INCLUDE PYTHON_CONFIG PYTHON_LIBRARY)
    if(NOT EXISTS "${SYNCWATCH_${name}}")
        message(FATAL_ERROR "Missing owned ${name}")
    endif()
endforeach()

add_library(cffi_backend MODULE "${SYNCWATCH_CFFI_SOURCE}/src/c/_cffi_backend.c")
set_target_properties(cffi_backend PROPERTIES PREFIX "" OUTPUT_NAME "_cffi_backend"
    C_STANDARD 11 C_STANDARD_REQUIRED YES C_VISIBILITY_PRESET hidden
    BUILD_WITH_INSTALL_RPATH TRUE)
target_compile_definitions(cffi_backend PRIVATE FFI_BUILDING=1)
target_include_directories(cffi_backend PRIVATE "${SYNCWATCH_PYTHON_INCLUDE}" "${SYNCWATCH_PYTHON_CONFIG}")
target_link_libraries(cffi_backend PRIVATE "${SYNCWATCH_PYTHON_LIBRARY}")
if(MSVC)
    enable_language(ASM_MASM)
    set(ffi "${SYNCWATCH_CFFI_SOURCE}/src/c/libffi_x86_x64")
    file(GLOB originals "${ffi}/*.c")
    list(REMOVE_ITEM originals "${ffi}/win32.c")
    target_sources(cffi_backend PRIVATE ${originals} "${ffi}/win64.asm")
    target_include_directories(cffi_backend PRIVATE "${ffi}")
    target_compile_options(cffi_backend PRIVATE "$<$<COMPILE_LANGUAGE:C>:/experimental:deterministic>")
    target_link_options(cffi_backend PRIVATE /Brepro)
    set_target_properties(cffi_backend PROPERTIES SUFFIX ".pyd")
else()
    if(NOT EXISTS "${SYNCWATCH_FFI_INCLUDE}/ffi.h" OR NOT EXISTS "${SYNCWATCH_FFI_LIBRARY}")
        message(FATAL_ERROR "Missing owned source-built libffi")
    endif()
    target_include_directories(cffi_backend PRIVATE "${SYNCWATCH_FFI_INCLUDE}")
    target_compile_definitions(cffi_backend PRIVATE USE__THREAD=1 HAVE_SYNC_SYNCHRONIZE=1)
    target_link_libraries(cffi_backend PRIVATE "${SYNCWATCH_FFI_LIBRARY}" ${CMAKE_DL_LIBS})
    if(APPLE)
        set_target_properties(cffi_backend PROPERTIES INSTALL_RPATH "@loader_path")
    else()
        set_target_properties(cffi_backend PROPERTIES INSTALL_RPATH "$ORIGIN")
    endif()
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/library-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:cffi_backend>")
