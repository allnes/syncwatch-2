cmake_minimum_required(VERSION 3.30)
project(SyncWatchPythonPackageProbe LANGUAGES C)
add_subdirectory("${SYNCWATCH_ROOT}/packages/python-package" package)
add_library(package_test SHARED exports.c)
target_link_libraries(package_test PRIVATE syncwatch_python_package)
if(MSVC)
    target_compile_options(package_test PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(package_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/module-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:package_test>")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/library-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:syncwatch_python_package>")
