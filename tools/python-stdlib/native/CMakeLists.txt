cmake_minimum_required(VERSION 3.30)
project(SyncWatchBootstrapStdlibProbe LANGUAGES C)
find_package(Threads REQUIRED)
add_subdirectory("${SYNCWATCH_ROOT}/packages/python-package" package)
add_executable(stdlib-probe probe.c "${SYNCWATCH_ROOT}/packages/python-startup/startup.c")
target_include_directories(stdlib-probe PRIVATE "${SYNCWATCH_FIXTURE}")
target_include_directories(stdlib-probe SYSTEM PRIVATE "${SYNCWATCH_PYTHON_INCLUDE}")
target_link_libraries(stdlib-probe PRIVATE syncwatch_python_package "${SYNCWATCH_PYTHON_LIBRARY}" Threads::Threads)
if(MSVC)
    target_compile_options(stdlib-probe PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(stdlib-probe PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/probe-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:stdlib-probe>")
