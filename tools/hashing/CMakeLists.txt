cmake_minimum_required(VERSION 3.30)
project(SyncWatchHaclHash LANGUAGES C)
set(names MD5 SHA1 SHA2 SHA3)
foreach(name IN LISTS names)
    list(APPEND sources "${SYNCWATCH_SOURCE}/hacl-star/dist/gcc-compatible/Hacl_Hash_${name}.c")
endforeach()
add_library(syncwatch_hacl STATIC ${sources})
target_compile_features(syncwatch_hacl PUBLIC c_std_11)
target_compile_definitions(syncwatch_hacl PUBLIC KRML_VERIFIED_UINT128)
target_include_directories(syncwatch_hacl SYSTEM PUBLIC "${SYNCWATCH_HEADERS}")
set_property(TARGET syncwatch_hacl PROPERTY POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    cmake_path(NATIVE_PATH CMAKE_BINARY_DIR NORMALIZE native_build)
    cmake_path(NATIVE_PATH SYNCWATCH_SOURCE NORMALIZE native_source)
    cmake_path(NATIVE_PATH SYNCWATCH_WORK NORMALIZE native_work)
    target_compile_options(syncwatch_hacl PRIVATE /W4 /WX /experimental:deterministic
        "/pathmap:${native_build}=syncwatch-build" "/pathmap:${native_source}=hash-source" "/pathmap:${native_work}=hash-work"
        "/FI${SYNCWATCH_HEADERS}/syncwatch_hacl_namespace.h")
    set_property(TARGET syncwatch_hacl APPEND PROPERTY STATIC_LIBRARY_OPTIONS /Brepro)
else()
    target_compile_options(syncwatch_hacl PRIVATE -Wall -Wextra -Werror
        "-ffile-prefix-map=${SYNCWATCH_SOURCE}=hash-source"
        "-ffile-prefix-map=${SYNCWATCH_WORK}=hash-work"
        -include "${SYNCWATCH_HEADERS}/syncwatch_hacl_namespace.h")
endif()
if(SYNCWATCH_ARTIFACT)
    add_library(accepted_hash STATIC IMPORTED)
    set_target_properties(accepted_hash PROPERTIES IMPORTED_LOCATION "${SYNCWATCH_ARTIFACT}")
    set(probe_library accepted_hash)
else()
    set(probe_library syncwatch_hacl)
endif()
add_executable(hash_probe probe.c)
target_compile_features(hash_probe PRIVATE c_std_11)
target_include_directories(hash_probe SYSTEM PRIVATE "${SYNCWATCH_HEADERS}")
target_include_directories(hash_probe PRIVATE "${SYNCWATCH_FIXTURES}")
target_link_libraries(hash_probe PRIVATE ${probe_library})
if(MSVC)
    target_compile_options(hash_probe PRIVATE /W4 /WX)
else()
    target_compile_options(hash_probe PRIVATE -Wall -Wextra -Werror)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/library-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:syncwatch_hacl>")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/probe-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:hash_probe>")
